cmake_minimum_required(VERSION 3.24)
project(FARecorder VERSION 0.4.0 LANGUAGES CXX)
if(MSVC AND DEFINED FAR_MSVC_INCLUDE_PREFIX)
  # Project-local correction for Korean-only MSVC installations whose detected
  # /showIncludes prefix was decoded incorrectly by CMake.
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${FAR_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS" FORCE)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Multimedia Concurrent)
qt_standard_project_setup(REQUIRES 6.8)

add_library(luma_core STATIC
  app/core/Types.h app/core/Platform.h
  app/core/SessionTimeline.h app/core/AudioCheck.h app/core/AudioCheck.cpp
  app/core/WaveWriter.h app/core/WaveWriter.cpp
  app/core/RecordingSession.h app/core/RecordingSession.cpp)
target_include_directories(luma_core PUBLIC app)
target_link_libraries(luma_core PUBLIC Qt6::Core Qt6::Gui)
if(WIN32)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/cppwinrt/include/winrt/base.h")
    target_include_directories(luma_core BEFORE PUBLIC .deps/cppwinrt/include)
  endif()
  target_sources(luma_core PRIVATE app/platform/windows/WindowsPlatform.cpp)
  target_compile_definitions(luma_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(luma_core PRIVATE d3d11 dxgi windowsapp ole32 uuid mmdevapi avrt dwmapi)
elseif(APPLE)
  target_sources(luma_core PRIVATE app/platform/macos/MacPlatform.mm)
  set_source_files_properties(app/platform/macos/MacPlatform.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(luma_core PRIVATE "-framework ScreenCaptureKit" "-framework AVFoundation" "-framework CoreMedia" "-framework CoreAudio" "-framework CoreVideo" "-framework CoreGraphics" "-framework AppKit")
else()
  message(FATAL_ERROR "FARecorder supports Windows and macOS.")
endif()

qt_add_executable(Luma WIN32 MACOSX_BUNDLE app/main.cpp app/RecorderController.h app/RecorderController.cpp app/GlobalHotkeys.h app/GlobalHotkeys.cpp)
set_target_properties(Luma PROPERTIES OUTPUT_NAME FARecorder)
target_link_libraries(Luma PRIVATE luma_core Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Concurrent)
set_source_files_properties(app/qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(Luma URI Luma VERSION 1.0 QML_FILES app/qml/Main.qml app/qml/Theme.qml app/qml/IconButton.qml app/qml/QuietCombo.qml app/qml/QuietMenu.qml app/qml/QuietDialog.qml app/qml/QuietSeparator.qml app/qml/QuietCheckBox.qml app/qml/AudioCard.qml app/qml/Panel.qml app/qml/ActionButton.qml app/qml/Field.qml app/qml/Meter.qml app/qml/AudioRoute.qml app/qml/TransportButton.qml)
target_sources(Luma PRIVATE app/SourceWatcher.h)
if(WIN32)
  target_sources(Luma PRIVATE app/SourceWatcher.cpp)
  target_compile_definitions(Luma PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  target_sources(Luma PRIVATE app/SourceWatcher.mm)
  set_source_files_properties(app/SourceWatcher.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
endif()
qt_add_resources(Luma artwork PREFIX "/" BASE app FILES app/assets/app-icon.png app/assets/record.png app/assets/stop.png app/assets/pause.png app/assets/resume.png)
qt_add_resources(Luma fonts PREFIX "/fonts" BASE app/assets/fonts FILES app/assets/fonts/D2Coding-Regular.ttf app/assets/fonts/D2Coding-Bold.ttf app/assets/fonts/OFL.txt)
qt_add_resources(Luma licenses PREFIX "/licenses" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES LICENSE THIRD_PARTY_NOTICES.md)
if(WIN32)
  target_sources(Luma PRIVATE app/app.rc)
endif()
if(MSVC)
  target_compile_options(luma_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(Luma PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
if(APPLE)
  target_link_libraries(Luma PRIVATE "-framework Carbon")
  target_sources(Luma PRIVATE app/assets/FARecorder.icns)
  set_source_files_properties(app/assets/FARecorder.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(Luma PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist" MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.yanggogi1423.farecorder")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(luma_tests tests/CoreTests.cpp)
  target_link_libraries(luma_tests PRIVATE luma_core)
  add_test(NAME core COMMAND luma_tests)
  add_test(NAME qml_smoke COMMAND Luma --ui-smoke)
  add_test(NAME qml_expanded COMMAND Luma --ui-smoke --expanded)
  add_test(NAME qml_custom COMMAND Luma --ui-smoke --custom)
  add_test(NAME qml_small COMMAND Luma --ui-smoke --small)
  add_test(NAME qml_audio COMMAND Luma --ui-smoke --populated)
  add_test(NAME qml_audio_expanded COMMAND Luma --ui-smoke --populated --expanded)
  add_test(NAME source_discovery COMMAND Luma --discovery-test)
  add_test(NAME window_controls COMMAND Luma --window-test)
  add_test(NAME settings_migration COMMAND Luma --settings-test)
  set_tests_properties(qml_smoke qml_expanded qml_custom qml_small qml_audio qml_audio_expanded source_discovery settings_migration window_controls PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
  if(WIN32)
    add_test(NAME encoder_preflight COMMAND Luma --encoder-test)
    set_tests_properties(encoder_preflight PROPERTIES TIMEOUT 25 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=software")
  endif()
endif()
install(TARGETS Luma BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES app/assets/fonts/OFL.txt DESTINATION licenses/D2Coding)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
if(APPLE)
  install(FILES app/assets/fonts/OFL.txt DESTINATION FARecorder.app/Contents/Resources/licenses/D2Coding)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION FARecorder.app/Contents/Resources/licenses)
endif()
qt_generate_deploy_qml_app_script(TARGET Luma OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
